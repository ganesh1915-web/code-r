def bubble_sort(salary):
    n = len(salary)
    for i in range(n-1):
        for j in range(n-i-1):
            if (salary[j]>salary[j+1]):
                salary[j] , salary[j+1] = salary[j+1] , salary[j]
                print(f" ",salary[j])
    return salary
