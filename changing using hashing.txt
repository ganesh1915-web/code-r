class linearProbingHashingTable:
    def __init__(self, size=10):
        self.size = size
        self.table = [None] * size
        self.DELETED = "<DELETED>"

    def hash_function(self, key):
        return key % self.size

    def insert(self, key):
        index = self.hash_function(key)
        original_index = index
        while self.table[index] not in (None, self.DELETED):
            if self.table[index] == key:
                print(f"key {key} already exists at index {index}.")
                return
            index = (index + 1) % self.size
            if index == original_index:
                print(f"Hash table is full. Cannot insert.")
                return
        self.table[index] = key
        print(f"Inserted key {key} at index {index}.")

    def search(self, key):
        index = self.hash_function(key)
        original_index = index
        while self.table[index] is not None:
            if self.table[index] == key:
                print(f"key {key} found at index {index}.")
                return index
            index = (index + 1) % self.size
            if index == original_index:
                break
        print(f"key {key} not found.")
        return None

    def delete(self, key):
        index = self.search(key)
        if index is not None:
            self.table[index] = self.DELETED
            print(f"key {key} is deleted from index {index}.")

    def display(self):
        print("Hash Table:")
        for i, key in enumerate(self.table):
            print(f"index {i}: {key}")

    def menu(self):
        while True:
            print("\n=== Menu ===")
            print("1. Insert a key")
            print("2. Search a key")
            print("3. Delete a key")
            print("4. Display the table")
            print("5. Exit")
            choice = int(input("Enter your choice: "))

            if choice == 1:
                key = int(input("Enter the key: "))
                self.insert(key)
            elif choice == 2:
                key = int(input("Enter key to search: "))
                self.search(key)
            elif choice == 3:
                key = int(input("Enter key to delete: "))
                self.delete(key)
            elif choice == 4:
                self.display()
            elif choice == 5:
                print("Exit")
                break
            else:
                print("Invalid choice. Try again.")


ht = linearProbingHashingTable()
ht.menu()
